% ==================== MAP-ELITES ====================
\section{MAP-Elites}

\begin{frame}{MAP-Elites: Koncepcja}
\begin{block}{Quality-Diversity}
Maintain archive of \textbf{elite solutions} across behavior space
\end{block}

\textbf{Archive: 6×6 grid (36 cells)}

\textbf{Behavior dimensions:}
\begin{align*}
\text{Gap Tightness} &= \frac{11 - \text{avg\_gap}}{5} \in [0,1] \\
\text{Spacing Density} &= \frac{52 - \text{avg\_spacing}}{17} \in [0,1]
\end{align*}

\vspace{0.3cm}
\begin{itemize}
\item Każda komórka: najlepszy poziom dla danego zachowania
\item Mutacja: rate=0.45, $\sigma$=0.25
\item Inicjalizacja: 250 random genomes
\end{itemize}
\end{frame}

\begin{frame}{MAP-Elites: Algorytm}
\begin{algorithmic}[1]
\STATE Initialize archive with 250 random genomes
\FOR{$i = 1$ \textbf{to} $2000$}
    \STATE parent $\leftarrow$ archive.get\_random\_elite()
    \STATE offspring $\leftarrow$ parent.mutate(rate=0.45, $\sigma$=0.25)
    \STATE quality, behavior $\leftarrow$ evaluate(offspring)
    \STATE cell $\leftarrow$ get\_cell\_index(behavior)
    \IF{archive[cell] is None \textbf{OR} quality $>$ archive[cell].quality}
        \STATE archive[cell] $\leftarrow$ offspring
    \ENDIF
\ENDFOR
\end{algorithmic}

\vspace{0.3cm}
\textbf{Kluczowe:} Każda iteracja potencjalnie wypełnia jedną komórkę
\end{frame}

\begin{frame}{MAP-Elites: Quality Metrics}
\begin{equation}
Q = 0.35 \cdot P + 0.35 \cdot B + 0.30 \cdot R - P_{flat} - P_{vert}
\end{equation}

\textbf{Komponenty:}
\begin{itemize}
\item $P$ (Playability): \textasciitilde50\% survival, progress, no instant death
\item $B$ (Balance): Bot performance ratio (ideal: 1.0-1.2$\times$)
\item $R$ (Progression): Smooth difficulty curve
\item $P_{flat}$ (Flat bot penalty): Wykrywa "flat" levels
\item $P_{vert}$ (Vertical penalty): Zbyt duża wariancja wysokości
\end{itemize}

\vspace{0.3cm}
\begin{block}{Range}
Quality $\in$ [-1.0, 1.0] (praktycznie: 0.3-0.8)
\end{block}
\end{frame}

\begin{frame}{MAP-Elites: Wyniki}
\textbf{Statystyki:}
\begin{itemize}
\item Coverage: \textbf{60-80\%} (24-29/36 cells)
\item Quality range: 0.3-0.8
\item Max quality: \textbf{0.73}
\item Iterations: 2000
\item Time: \textasciitilde5 min
\end{itemize}

\vspace{0.5cm}
\begin{exampleblock}{Zalety}
\begin{itemize}
\item Gwarantuje diversity (pełne pokrycie behavior space)
\item Wysokie max quality
\item Prosty i niezawodny
\end{itemize}
\end{exampleblock}

\begin{alertblock}{Wady}
Powolna konwergencja (2000 iteracji)
\end{alertblock}
\end{frame}

% ==================== MCTS-QD ====================
\section{MCTS-QD}

\begin{frame}{MCTS-QD: Koncepcja}
\textbf{Monte Carlo Tree Search + Quality-Diversity Archive}

\begin{block}{State-Action Space}
\begin{itemize}
\item \textbf{State:} partial level (pipes built so far)
\item \textbf{Action:} add next pipe (gap\_size, gap\_center, spacing, items)
\item \textbf{Actions per state:} 6 × 5 × 5 × 4 = \textbf{600}
\end{itemize}
\end{block}

\textbf{Parametry:}
\begin{itemize}
\item Simulations per level: 200-300
\item Exploration constant: $C = 1.414$ ($\sqrt{2}$)
\item Target pipes: 17 (range: 15-19)
\end{itemize}

\vspace{0.3cm}
\centering
\textit{Tree search $\rightarrow$ lepsze próbki $\rightarrow$ sample efficiency}
\end{frame}

\begin{frame}{MCTS-QD: UCT Formula}
\begin{equation}
\text{UCT}(node) = \underbrace{\frac{Q}{N}}_{\text{exploit}} + C\underbrace{\sqrt{\frac{\ln N_{parent}}{N}}}_{\text{explore}}
\end{equation}

where:
\begin{itemize}
\item $Q$ = total reward
\item $N$ = visit count
\item $C = 1.414$ (exploration constant)
\end{itemize}

\vspace{0.5cm}
\begin{block}{Balans}
\begin{itemize}
\item \textbf{Exploitation:} Wybierz akcje z wysokim average reward
\item \textbf{Exploration:} Eksploruj rzadziej odwiedzane akcje
\end{itemize}
\end{block}
\end{frame}

\begin{frame}[fragile]{MCTS-QD: Fazy Algorytmu}
\textbf{4 fazy (dla każdej symulacji):}

\begin{enumerate}
\item \textbf{SELECT:} Traverse tree via highest UCT
\item \textbf{EXPAND:} Add new child node
\item \textbf{SIMULATE:} Random rollout to completion
\item \textbf{BACKPROPAGATE:} Update rewards up tree
\end{enumerate}

\vspace{0.5cm}
\lstinputlisting{code/uct_selection.py}

\vspace{0.3cm}
Po 200-300 symulacjach: trace best path $\rightarrow$ genome
\end{frame}

\begin{frame}{MCTS-QD: Wyniki vs MAP-Elites}
\begin{table}
\centering
\begin{tabular}{lcc}
\toprule
\textbf{Metryka} & \textbf{MAP-Elites} & \textbf{MCTS-QD} \\
\midrule
Coverage & 60-80\% & 30-50\% \\
Max Quality & 0.73 & \textbf{0.62} \\

\end{block}

% ==================== NOVELTY SEARCH ====================
\section{Novelty Search}

\begin{frame}{Novelty Search: Koncepcja}
\begin{alertblock}{Kluczowa Różnica}
\textbf{Fitness = Novelty} (not Quality!)
\end{alertblock}

\begin{equation}
\text{novelty}(x) = \frac{1}{k}\sum_{i=1}^{k} \text{dist}(x, \text{neighbor}_i)
\end{equation}

\textbf{k-NN distance in behavior space} (k=15)

\vspace{0.5cm}
\textbf{Archive:}
\begin{itemize}
\item \textbf{Struktura:} List (not grid), max 1000 individuals
\item \textbf{Selection:} Najmniej nowatorski zastępowany przez bardziej nowatorski
\item \textbf{Quality:} Zapisywana, ale nie optymalizowana
\end{itemize}
\end{frame}

\begin{frame}{Novelty Search: Algorytm}
\begin{algorithmic}[1]
\STATE Initialize: 200 random genomes
\FOR{$i = 1$ \textbf{to} $1000$}
    \STATE parent $\leftarrow$ archive.get\_random()
    \STATE offspring $\leftarrow$ parent.mutate()
    \STATE behavior $\leftarrow$ evaluate(offspring)
    \STATE novelty $\leftarrow$ k\_NN\_distance(behavior, k=15)
    \IF{archive.size $<$ 1000 \textbf{OR} novelty $>$ min\_novelty}
        \STATE archive.add(offspring)
    \ENDIF
\ENDFOR
\end{algorithmic}

\vspace{0.5cm}
\textbf{Kluczowe:} Brak pressure na quality $\rightarrow$ pure exploration
\end{frame}

\begin{frame}[fragile]{Novelty Search: k-NN Implementation}
\lstinputlisting{code/novelty_knn.py}

\vspace{0.5cm}
\textbf{Intuicja:}
\begin{itemize}
\item Wysoka novelty = daleko od istniejących poziomów
\item Promuje diverse behavior patterns
\item Może odkryć niespodziewane, wysokiej jakości struktury
\end{itemize}
\end{frame}

\begin{frame}{Novelty Search: Wyniki}
\textbf{Statystyki:}
\begin{itemize}
\item Archive size: 500-1000 individuals
\item Behavior spread: \textbf{wysoka diversity}
\item Quality: 0.2-0.8 (wide variance)
\item Max quality: 0.79 (przypadkowy spike)
\item Iterations: 1000
\end{itemize}

\vspace{0.5cm}
\begin{exampleblock}{Zalety}
\begin{itemize}
\item \textbf{Najwyższa diversity} - odkrywa nietypowe struktury
\item Brak local optima w quality space
\item Może znaleźć surprising solutions
\end{itemize}
\end{exampleblock}

\begin{alertblock}{Wady}
Brak gwarancji quality - niektóre poziomy niskiej jakości
\end{alertblock}
\end{frame}
